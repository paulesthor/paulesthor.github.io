Étape 1 : Fusion ETIM et PIM (Archivage)

Note : Cette opération est désormais gérée nativement dans l'onglet 'Analyse en Masse'. Ce script Python est conservé ici pour archivage.

import pandas as pd
import time
import os
import tkinter as tk
from tkinter import filedialog

def preparer_fichier_web():
    root = tk.Tk()
    root.attributes('-topmost', True)
    root.withdraw()

    print("DÉMARRAGE DE LA FUSION (MODE MULTI-ETIM)...")

    print("\n[1/4] En attente de la sélection des fichiers ETIM...")
    fichiers_etim = filedialog.askopenfilenames(
        title="1. Sélectionner les fichiers ETIM (Format : Technical data per line)",
        filetypes=[("Fichiers Excel", "*.xlsx *.xls")]
    )
    
    if not fichiers_etim: return
    
    print("\n[2/4] En attente de la sélection du fichier PIM géant...")
    fichier_pim = filedialog.askopenfilename(
        title="2. Sélectionner le fichier PIM (Format : Current view)",
        filetypes=[("Fichiers Excel", "*.xlsx *.xls")]
    )
    
    if not fichier_pim: return

    dossier_sortie = os.path.dirname(fichier_pim)
    fichier_sortie = os.path.join(dossier_sortie, "Produits_Prets_Pour_IA.xlsx")
    start_time = time.time()

    liste_df_etim = []
    for f in fichiers_etim:
        temp_df = pd.read_excel(f, dtype=str)
        liste_df_etim.append(temp_df)
    
    df_etim = pd.concat(liste_df_etim, ignore_index=True)
    
    df_etim['INFO_COMPLETE'] = df_etim['FEATUREID'] + ": " + df_etim['FVALUE']
    df_etim_groupe = df_etim.groupby('MARQUE')['INFO_COMPLETE'].apply(lambda x: ' | '.join(x.dropna())).reset_index()
    df_etim_groupe.rename(columns={'INFO_COMPLETE': 'SPECIFICATIONS_ETIM'}, inplace=True)
    
    ids_valides = df_etim_groupe['MARQUE'].unique()

    df_pim = pd.read_excel(fichier_pim, dtype=str)
    df_pim_filtre = df_pim[df_pim[''].isin(ids_valides)].copy()
    df_final = pd.merge(df_pim_filtre, df_etim_groupe, left_on='', right_on='MARQUE', how='inner')

    df_final = df_final[df_final['Code Douane'].notna()]
    df_final = df_final[df_final['Code Douane'].astype(str).str.strip() != ""]

    df_export = df_final[['Code Douane', '', 'LFLG0254- Description (80 car) (EN)', 'Class Legrand', 'SPECIFICATIONS_ETIM']].copy()
    df_export.rename(columns={'': 'ID_PRODUIT', 'LFLG0254- Description (80 car) (EN)': 'DESCRIPTION_COMMERCIALE', 'Class Legrand': 'CLASS LEGRAND'}, inplace=True)
    df_export.to_excel(fichier_sortie, index=False)

    print(f"TERMINÉ en {round(time.time() - start_time, 1)} secondes !")

if __name__ == "__main__":
    preparer_fichier_web()

Étape 2 : Entraînement Modèle Encyclopédie (Seuil 3)

Génère le modèle d'analyse large. Nécessite le fichier Excel fusionné issu de l'étape 1.

import pandas as pd
import numpy as np
import json
import zipfile
import re
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import LabelEncoder
from sklearn.utils.class_weight import compute_sample_weight
import xgboost as xgb
import onnxmltools
from onnxmltools.convert.common.data_types import FloatTensorType

FICHIER_ENTRAINEMENT = "Produits_Prets_Pour_IA.xlsx"
MAX_FEATURES = 5000 

def pipeline_entrainement_encyclopedie():
    df = pd.read_excel(FICHIER_ENTRAINEMENT, dtype=str).fillna("")
    col_douane = [c for c in df.columns if 'DOUANE' in c.upper() or 'SH' in c.upper()][0]
    
    comptage = df[col_douane].value_counts()
    df = df[df[col_douane].isin(comptage[comptage >= 3].index)]

    def preparer_texte(row):
        etim_boost = row.get('SPECIFICATIONS_ETIM', '').lower()
        for c in ['material', 'matiere', 'thermoplastic', 'voltage', 'tension', '230v', 'current', 'courant', 'ampere', 'pole', 'mounting', 'type']:
            if c in etim_boost: etim_boost += f" {c} {c} {c}"
        texte_brut = f"class {row.get('CLASS LEGRAND', '')} class {row.get('CLASS LEGRAND', '')} desc {row.get('DESCRIPTION_COMMERCIALE', '')} {etim_boost}".lower()
        return re.sub(r'[^\w\s]', ' ', texte_brut)

    df['TEXTE_ML'] = df.apply(preparer_texte, axis=1)

    vectorizer = TfidfVectorizer(max_features=MAX_FEATURES, ngram_range=(1, 2), token_pattern=r"(?u)\b\w+\b")
    X_dense = vectorizer.fit_transform(df['TEXTE_ML']).toarray() 
    
    le = LabelEncoder()
    y = le.fit_transform(df[col_douane])
    X_train, X_test, y_train, y_test = train_test_split(X_dense, y, test_size=0.2, random_state=42, stratify=y)

    modele = xgb.XGBClassifier(objective='multi:softprob', eval_metric='mlogloss', tree_method='hist', n_estimators=100, learning_rate=0.1, max_delta_step=1, early_stopping_rounds=10, random_state=42, n_jobs=-1)
    modele.fit(X_train, y_train, sample_weight=compute_sample_weight('balanced', y=y_train), eval_set=[(X_train, y_train), (X_test, y_test)], verbose=10)

    onx = onnxmltools.convert_xgboost(modele, initial_types=[('float_input', FloatTensorType([None, MAX_FEATURES]))], target_opset=12)
    with open("modele_encyclopedie.onnx", "wb") as f: f.write(onx.SerializeToString())
    
    with open('config_encyclopedie.json', 'w') as f: json.dump({'vocabulaire': {str(k): int(v) for k, v in vectorizer.vocabulary_.items()}, 'idf': vectorizer.idf_.tolist(), 'codes_sh': le.classes_.astype(str).tolist()}, f)

if __name__ == "__main__": pipeline_entrainement_encyclopedie()
